#     SIRIUS_LLM_ALLOWED_HOSTS 追加；云元数据地址(169.254.x)一律拒绝。

SDTM_MAX_WORKERS=5

# ================================
# 级联预测策略 (Cascade Prediction)
# ================================
# 4级级联策略，高置信度时跳过LLM调用以降低成本和延迟
#
# Level 1: KB 精确匹配 (confidence >= KB_MIN_CONFIDENCE) → 直接采纳
# Level 2: KB 高置信度 (confidence >= CASCADE_KB_HIGH_CONF) → 直接采纳
# Level 3: RAG 增强检索 (top_score >= CASCADE_RAG_HIGH_CONF) → 直接采纳
# Level 4: LLM 推理 (以上均不满足时) → 调用LLM

# KB 最小置信度阈值 (Level 1 安全下限，需同时满足 source=eCRF_direct_match)
KB_MIN_CONFIDENCE=0.8

# KB 高置信度阈值 (Level 2 退出阈值，KB建议直接采纳)
CASCADE_KB_HIGH_CONF=0.85

# RAG 高置信度阈值 (Level 3 退出阈值，RAG结果直接采纳)
CASCADE_RAG_HIGH_CONF=0.7

# KB 域覆盖置信度 (超过此阈值时允许覆盖推断域)
KB_DOMAIN_OVERRIDE_CONF=0.85

# 批处理大小
BATCH_SIZE=10

# LLM API 调用频率限制 (请求/分钟)
# 并行模式下使用令牌桶算法进行线程安全的速率控制
